cmake_minimum_required(VERSION 3.18)
project(OpenPLLKLSProfile LANGUAGES CXX)
find_package(Trilinos REQUIRED)
set(KLS_ROOT /tmp/openpll-kls-detailed-20261025-001)
set(wraps kls_analyze_csr kls_factor kls_refactor kls_solve kls_solve_transpose kls_refactor_solve kls_componentwise_certify_correction kls_componentwise_certify_verdict_correction trilinos_klu_l_factor trilinos_klu_l_refactor trilinos_klu_l_solve trilinos_klu_l_tsolve malloc calloc realloc free aligned_alloc posix_memalign)
foreach(flavor normal coarse sample)
 foreach(kind rhs matrix)
  set(target compare_${flavor}_${kind})
  add_executable(${target} ${kind}.cpp)
  target_compile_features(${target} PRIVATE cxx_std_17)
  target_compile_options(${target} PRIVATE -g)
  target_include_directories(${target} PRIVATE ${Trilinos_INCLUDE_DIRS} ${Trilinos_TPL_INCLUDE_DIRS} ${KLS_ROOT}/KLS/include)
  target_link_directories(${target} PRIVATE ${Trilinos_LIBRARY_DIRS} ${Trilinos_TPL_LIBRARY_DIRS})
  if(flavor STREQUAL "coarse")
   set(kls_library ${KLS_ROOT}/libkls_coarse.a)
   target_link_libraries(${target} PRIVATE ${KLS_ROOT}/phase.o ${KLS_ROOT}/cpu.o)
   foreach(name ${wraps})
    target_link_options(${target} PRIVATE -Wl,--wrap=${name})
   endforeach()
  else()
   set(kls_library ${KLS_ROOT}/build/libkls.a)
   if(flavor STREQUAL "sample")
    target_link_libraries(${target} PRIVATE ${KLS_ROOT}/cpu.o)
   endif()
  endif()
  target_link_libraries(${target} PRIVATE ${Trilinos_LIBRARIES} ${Trilinos_TPL_LIBRARIES}
   ${kls_library} ${KLS_ROOT}/build/libkls_suitesparse.a
   ${KLS_ROOT}/build/libkls_metis.a ${KLS_ROOT}/build/libkls_gklib.a
   ${KLS_ROOT}/build/lib/libscotch.a ${KLS_ROOT}/build/lib/libscotcherr.a
   ${KLS_ROOT}/build/libkls_spral_scaling.a gfortran quadmath gomp dl)
 endforeach()
endforeach()
