cmake_minimum_required(VERSION 3.16)
project(OpenPLLMatvecProbe LANGUAGES CXX)
find_package(Trilinos REQUIRED)
find_package(OpenMP REQUIRED)
add_executable(native_matvec_probe main.cpp)
target_compile_features(native_matvec_probe PRIVATE cxx_std_17)
target_include_directories(native_matvec_probe PRIVATE ${Trilinos_INCLUDE_DIRS} ${Trilinos_TPL_INCLUDE_DIRS})
target_link_directories(native_matvec_probe PRIVATE ${Trilinos_LIBRARY_DIRS} ${Trilinos_TPL_LIBRARY_DIRS})
target_link_libraries(native_matvec_probe PRIVATE ${Trilinos_LIBRARIES} ${Trilinos_TPL_LIBRARIES} OpenMP::OpenMP_CXX)
